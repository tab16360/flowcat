cmake_minimum_required(VERSION 3.24)
project(FlowCatWindows VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_executable(FlowCat WIN32 src/main.cpp src/collector.cpp src/store.cpp src/health.cpp app.rc)
target_link_options(FlowCat PRIVATE /MANIFEST:NO)
target_compile_options(FlowCat PRIVATE /utf-8 /W4 /EHsc)
set_property(TARGET FlowCat PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(FlowCat PRIVATE advapi32 tdh iphlpapi ws2_32 wlanapi shell32 shlwapi comctl32 d2d1 dwrite dwmapi ole32 psapi comdlg32 uxtheme)
add_executable(FlowCatCoreTests tests/core.cpp)
target_compile_options(FlowCatCoreTests PRIVATE /utf-8 /W4 /EHsc)
set_property(TARGET FlowCatCoreTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_test(NAME core COMMAND FlowCatCoreTests)
add_executable(FlowCatStoreTests tests/store.cpp src/store.cpp)
target_compile_options(FlowCatStoreTests PRIVATE /utf-8 /W4 /EHsc)
set_property(TARGET FlowCatStoreTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(FlowCatStoreTests PRIVATE shell32 ole32)
add_test(NAME persistence COMMAND FlowCatStoreTests)
add_executable(FlowCatUiCapture WIN32 tests/ui_capture.cpp)
target_compile_options(FlowCatUiCapture PRIVATE /utf-8 /W4 /EHsc)
set_property(TARGET FlowCatUiCapture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(FlowCatUiCapture PRIVATE ole32 windowscodecs shell32)
